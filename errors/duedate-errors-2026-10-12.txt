import re

def analyze_netsuite_log(log_content):
    """
    Analyzes NetSuite log content to find SCRIPT ERRORs and extract specific details.

    Args:
        log_content (str): The full content of the NetSuite log.

    Returns:
        list: A list of dictionaries, each representing an extracted error,
              or an empty list if no errors are found.
    """
    errors = []
    lines = log_content.strip().split('\n')
    
    # Regex to find SCRIPT ERROR lines and capture their timestamp
    script_error_pattern = re.compile(r"\[(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3})\] SCRIPT ERROR:")
    
    # Regex to extract parameters from the "Parameters Used:" line and the line after it
    param_amount_pattern = re.compile(r"billPaymentAmount: (\d+)")
    param_utr_date_pattern = re.compile(r"utrGeneratedDate: (\S+)")
    param_bill_id_pattern = re.compile(r"billId: (\d+)")
    param_invoice_id_pattern = re.compile(r"invoiceId: (\d+)")

    i = 0
    while i < len(lines):
        line = lines[i]
        match = script_error_pattern.match(line)

        if match:
            error_timestamp = match.group(1)
            current_error = {
                "timestamp": error_timestamp,
                "Type": "Script Error",
                "Bill Payment ID": "N/A",
                "Invoice ID": "N/A",
                "Bill ID": "N/A",
                "Amount": "N/A",
                "UTR Date": "N/A",
                "Error Name": "N/A",
                "Error Message": "N/A"
            }
            
            # Start looking for error details from the next line
            j = i + 1
            while j < len(lines):
                detail_line = lines[j].strip()

                # Stop processing this error block if we hit another timestamped line
                # or the "Stack:" line (which marks the end of the structured error details)
                if re.match(r"\[\d{4}-\d{2}-\d{2}", detail_line) or detail_line.startswith("Stack:"):
                    break

                if detail_line.startswith("Bill Payment ID:"):
                    current_error["Bill Payment ID"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Error Name:"):
                    current_error["Error Name"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Error Message:"):
                    current_error["Error Message"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Parameters Used:"):
                    # Extract Amount and UTR Date from this line
                    amount_match = param_amount_pattern.search(detail_line)
                    if amount_match:
                        current_error["Amount"] = amount_match.group(1)
                    utr_date_match = param_utr_date_pattern.search(detail_line)
                    if utr_date_match:
                        current_error["UTR Date"] = utr_date_match.group(1)
                elif "billId:" in detail_line and "invoiceId:" in detail_line:
                    # This line often follows "Parameters Used:" and contains Bill ID and Invoice ID
                    bill_id_match = param_bill_id_pattern.search(detail_line)
                    if bill_id_match:
                        current_error["Bill ID"] = bill_id_match.group(1)
                    invoice_id_match = param_invoice_id_pattern.search(detail_line)
                    if invoice_id_match:
                        current_error["Invoice ID"] = invoice_id_match.group(1)
                
                j += 1
            
            errors.append(current_error)
            i = j # Continue scanning from where the error block ended
        else:
            i += 1 # Move to the next line if no SCRIPT ERROR found

    return errors
