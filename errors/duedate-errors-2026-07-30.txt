def parse_netsuite_errors(log_content):
    errors = []
    log_lines = log_content.splitlines()
    error_count = 0

    i = 0
    while i < len(log_lines):
        line = log_lines[i]
        if "SCRIPT ERROR:" in line:
            error_count += 1
            error_details = {
                "timestamp": line.split(']')[0][1:],
                "type": "Script Error",
                "bill_payment_id": "N/A",
                "invoice_id": "N/A",
                "bill_id": "N/A",
                "amount": "N/A",
                "utr_date": "N/A",
                "error_name": "N/A",
                "error_message": "N/A"
            }

            # Read subsequent lines for error details
            j = i + 1
            while j < len(log_lines):
                detail_line = log_lines[j].strip()

                if detail_line.startswith("Bill Payment ID:"):
                    error_details["bill_payment_id"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Error Name:"):
                    error_details["error_name"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Error Message:"):
                    error_details["error_message"] = detail_line.split(":", 1)[1].strip()
                elif detail_line.startswith("Parameters Used:"):
                    params_str = detail_line.split(":", 1)[1].strip()
                    
                    # Extract parameters using regex for robustness
                    bill_pay_id_match = re.search(r'billPayId:\s*(\d+)', params_str)
                    if bill_pay_id_match:
                        # This might overwrite the one from "Bill Payment ID:" line, but it's fine as they should be same
                        error_details["bill_payment_id"] = bill_pay_id_match.group(1) 
                    
                    amount_match = re.search(r'billPaymentAmount:\s*(\d+)', params_str)
                    if amount_match:
                        error_details["amount"] = amount_match.group(1)
                    
                    utr_date_match = re.search(r'utrGeneratedDate:\s*([\d/]+)', params_str)
                    if utr_date_match:
                        error_details["utr_date"] = utr_date_match.group(1)
                    
                    bill_id_match = re.search(r'billId:\s*(\d+)', params_str)
                    if bill_id_match:
                        error_details["bill_id"] = bill_id_match.group(1)
                    
                    invoice_id_match = re.search(r'invoiceId:\s*(\d+)', params_str)
                    if invoice_id_match:
                        error_details["invoice_id"] = invoice_id_match.group(1)
                elif detail_line.startswith("Stack:"):
                    # Stop reading details once stack trace starts or a new log entry begins
                    break
                elif re.match(r'^\[\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3}\]', detail_line):
                    # Stop if a new timestamped log line is encountered
                    break
                j += 1
            errors.append(error_details)
            i = j - 1 # Continue search from where we left off after processing this error
        i += 1
    
    if not errors:
        return "NO ERRORS DETECTED"
    
    formatted_output = []
    for idx, err in enumerate(errors):
        formatted_output.append(f"Error {idx + 1}: {err['timestamp']}")
        formatted_output.append(f"   Type: {err['type']}")
        formatted_output.append(f"   Bill Payment ID: {err['bill_payment_id']}")
        formatted_output.append(f"   Invoice ID: {err['invoice_id']}")
        formatted_output.append(f"   Bill ID: {err['bill_id']}")
        formatted_output.append(f"   Amount: {err['amount']}")
        formatted_output.append(f"   UTR Date: {err['utr_date']}")
        formatted_output.append(f"   Error Name: {err['error_name']}")
        formatted_output.append(f"   Error Message: {err['error_message']}")

    return "\n".join(formatted_output)

print(parse_netsuite_errors(log_content))
